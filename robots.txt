# We do not want this site in search results.
#
# Counter-intuitively, that means search crawlers are ALLOWED here. A
# "Disallow: /" only stops them fetching the page, and a URL they cannot fetch
# can still be listed as a bare link if anything else links to it. Keeping them
# out of the index requires them to actually read the noindex meta tag that
# every page carries, which means letting them fetch the page. Please do not
# "fix" this to Disallow: / — that would weaken it.
#
# AI crawlers are a different case: they do not index, they collect, so for
# them a flat refusal is the right answer. See also /ai.txt and
# /.well-known/tdmrep.json.

User-agent: *
Allow: /

# relying on each operator's reading of the wildcard above.
